# tejo — CLI local Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Publicar un único paquete npm `tejo` (bin `tejo`) con `init`, `mcp`, `view` y `check` sobre una carpeta local de conocimiento, y renombrar el proyecto a tejo.

**Architecture:** Los paquetes del monorepo pasan a `@tejo/*` (privados). `@tejo/mcp` deja de ser binario y exporta `FloFolder`, `createServer`, `INSTRUCTIONS` y `runMcp`. Un paquete nuevo `packages/tejo` contiene la CLI (`node:util.parseArgs`), `init`, `check` y el servidor `node:http` del visor; `vite build --ssr` lo empaqueta junto con core y mcp en `dist/cli.js`, y el build de `@tejo/web` se copia a `dist/viewer/`. La web detecta la API local (`api/files`) y entra en modo carpeta local.

**Tech Stack:** TypeScript (NodeNext en core/mcp/tejo), Vite 5 (build SSR), Vitest 2, pnpm 9, Node ≥20, `@modelcontextprotocol/sdk`.

**Spec:** `docs/superpowers/specs/2026-10-06-tejo-cli-design.md`

```ts
  /** La uri canónica y el archivo de `path`, o el error si sale de la raíz, no es .md/.flo (o una vista) o está ignorada. */
  private target(path: string, allowViews: boolean): { uri: string; file: string } | WriteResult {
    const uri = this.canonicalUri(path);
    if (uri === null) return fail(`la ruta '${path}' sale de la carpeta raíz (directamente o por un enlace simbólico)`);
    const view = allowViews && VIEW_URI.test(uri);
    if (!view && !isDocumentUri(uri)) return fail(`solo se escriben archivos .md o .flo: '${path}'`);
    if (isIgnoredUri(uri)) return fail(`la ruta '${path}' está en una carpeta ignorada (node_modules, .git)`);
    return { uri, file: join(this.root, uri) };
  }

  /** Escribe en un temporal y renombra; null si salió bien. */
  private writeAtomic(path: string, file: string, content: string): WriteResult | null {
    const temp = join(dirname(file), `.${basename(file)}.${randomBytes(6).toString("hex")}.tmp`);
    try {
      mkdirSync(dirname(file), { recursive: true });
      writeFileSync(temp, content);
      renameSync(temp, file);
      return null;
    } catch (error) {
      try {
        rmSync(temp, { force: true });
      } catch {
        // El temporal puede colgar de una ruta que ya no es un directorio (ENOTDIR); no hay nada que limpiar.
      }
      return fail(`no se pudo escribir '${path}': ${(error as NodeJS.ErrnoException).code ?? "error de E/S"}`);
    }
  }

  /** Copia del mapa `uri → fuente` (lo que sirve `GET /api/files` de `tejo view`). */
  files(): Record<string, string> {
    return Object.fromEntries(this.sources);
  }

  /** "Guardar" del visor (#66): mismas reglas de ruta que `write`, más las vistas; sin validar diagnósticos (#38). */
  save(path: string, content: string): WriteResult {
    const target = this.target(path, true);
    if (!("file" in target)) return target;
    const failed = this.writeAtomic(path, target.file, content);
    if (failed) return failed;
    if (isDocumentUri(target.uri)) {
      this.sources.set(target.uri, content);
      this.workspace.setDocument(target.uri, content);
      this.linked = null;
    }
    return { ok: true, uri: target.uri };
  }

import { createServer } from "./tools.js";

/** `tejo mcp`: servidor MCP por stdio sobre `root`. stdout es el protocolo: los avisos van a stderr. */
export async function runMcp(root: string): Promise<void> {
  const folder = new FloFolder(root);
  folder.load();
  folder.watch();
  await createServer(folder).connect(new StdioServerTransport());
  // El watcher mantiene vivo el proceso: sin esto, el servidor sobrevive al
  // cliente que cerró stdin.
  process.stdin.on("end", () => process.exit(0));
  process.stdin.on("close", () => process.exit(0));
}

import { statSync } from "node:fs";
import { resolve } from "node:path";

export const DEFAULT_FOLDER = "knowledge";

/** La carpeta de conocimiento de un comando, absoluta; falla si no es un directorio. */
export function resolveFolder(arg: string | undefined, cwd: string): string {
  const name = arg ?? DEFAULT_FOLDER;
  const path = resolve(cwd, name);
  if (!statSync(path, { throwIfNoEntry: false })?.isDirectory()) {
    throw new Error(`la carpeta '${name}' no existe o no es un directorio`);
  }
  return path;
}

import { formatDiagnostic } from "@tejo/core";
import { FloFolder } from "@tejo/mcp";

/** `tejo check` (spec §4.4): todo diagnóstico es un error (#53). */
export function check(root: string): { lines: string[]; errors: number } {
  const folder = new FloFolder(root);
  folder.load();
  const files = Object.keys(folder.files()).length;
  const diagnostics = folder.diagnostics() ?? {};
  const lines: string[] = [];
  for (const uri of Object.keys(diagnostics).sort()) {
    for (const diagnostic of diagnostics[uri]) lines.push(`${uri} ${formatDiagnostic(diagnostic)}`);
  }
  const withErrors = Object.keys(diagnostics).length;
  lines.push(lines.length === 0 ? `Sin errores en ${files} archivos.` : `${lines.length} errores en ${withErrors} archivos.`);
  return { lines, errors: lines.length - 1 };
}

import { realpathSync } from "node:fs";
import { fileURLToPath } from "node:url";
import { parseArgs } from "node:util";
import { runMcp } from "@tejo/mcp";
import { check } from "./check.js";
import { DEFAULT_FOLDER, resolveFolder } from "./folder-arg.js";

export const USAGE = `uso:
  tejo init [carpeta]                     prepara la carpeta, .mcp.json, la skill y AGENTS.md
  tejo mcp [carpeta]                      servidor MCP por stdio (lo arranca el agente)
  tejo view [carpeta] [--port N] [--no-open]  visor en http://127.0.0.1:8356
  tejo check [carpeta]                    lista los errores; sale con 1 si hay alguno

carpeta: por defecto '${DEFAULT_FOLDER}'.`;

/** Corre un comando y devuelve el exit code; los errores esperables salen como mensaje, no como excepción. */
export async function main(argv: string[], cwd: string): Promise<number> {
  const [command, ...rest] = argv;
  if (command === undefined || command === "--help" || command === "-h" || command === "help") {
    console.log(USAGE);
    return 0;
  }
  try {
    switch (command) {
      case "check": {
        const { positionals } = parseArgs({ args: rest, allowPositionals: true });
        const { lines, errors } = check(resolveFolder(positionals[0], cwd));
        for (const line of lines) (errors > 0 ? console.error : console.log)(line);
        return errors > 0 ? 1 : 0;
      }
      case "mcp": {
        const { positionals } = parseArgs({ args: rest, allowPositionals: true });
        await runMcp(resolveFolder(positionals[0], cwd));
        return 0;
      }
      default:
        console.error(`comando desconocido: ${command}\n\n${USAGE}`);
        return 1;
    }
  } catch (error) {
    console.error(`tejo: ${error instanceof Error ? error.message : String(error)}`);
    return 1;
  }
}

// Punto de entrada: realpath porque npx/npm llaman al bin por un enlace simbólico.
const entry = process.argv[1] ? realpathSync(process.argv[1]) : "";
if (entry === fileURLToPath(import.meta.url)) {
  const code = await main(process.argv.slice(2), process.cwd());
  // `mcp` (y `view`, Task 5) siguen vivos: no se fuerza la salida con código 0.
  if (code !== 0) process.exit(code);
}

import { existsSync, mkdirSync, readFileSync, readdirSync, writeFileSync } from "node:fs";
import { dirname, isAbsolute, join, relative, resolve, sep } from "node:path";
import { isDocumentUri } from "@tejo/core";
import { BLOCK_END, BLOCK_START, WELCOME_NOTE, agentsBlock, skill } from "./templates.js";

export type InitReport = { created: string[]; updated: string[]; unchanged: string[] };

type Write = { path: string; content: string };

const readIfExists = (file: string): string | null => (existsSync(file) ? readFileSync(file, "utf8") : null);

/** ¿La carpeta (recorrida sin seguir enlaces) tiene algún .flo o .md? */
function hasDocuments(dir: string): boolean {
  if (!existsSync(dir)) return false;
  return readdirSync(dir, { recursive: true, withFileTypes: false }).some((entry) => isDocumentUri(String(entry)));
}

/** `AGENTS.md` con el bloque de tejo puesto o reemplazado; el resto del texto no se toca. */
function withBlock(current: string | null, block: string): string {
  if (current === null) return block;
  const start = current.indexOf(BLOCK_START);
  const end = current.indexOf(BLOCK_END);
  if (start >= 0 && end > start) {
    const after = current.slice(end + BLOCK_END.length).replace(/^\n/, "");
    return current.slice(0, start) + block + after;
  }
  return `${current.replace(/\n*$/, "\n")}\n${block}`;
}

/** `tejo init` (spec §4.1, decisión #65): idempotente; calcula todo antes de escribir. */
export function init(cwd: string, folderArg: string): InitReport {
  const absolute = resolve(cwd, folderArg);
  const rel = relative(cwd, absolute);
  if (rel === "" || rel === ".." || rel.startsWith(`..${sep}`) || isAbsolute(rel)) {
    throw new Error(`la carpeta tiene que estar dentro del proyecto: '${folderArg}'`);
  }
  const folder = rel.split(sep).join("/");

  const mcpPath = join(cwd, ".mcp.json");
  const mcpText = readIfExists(mcpPath);
  let config: Record<string, unknown> = {};
  if (mcpText !== null) {
    try {
      config = JSON.parse(mcpText);
    } catch (error) {
      throw new Error(`.mcp.json no es JSON válido: ${(error as Error).message}`);
    }
  }
  const servers = (config.mcpServers ?? {}) as Record<string, unknown>;
  config.mcpServers = { ...servers, tejo: { command: "npx", args: ["tejo", "mcp", folder] } };

  const writes: Write[] = [
    { path: ".mcp.json", content: `${JSON.stringify(config, null, 2)}\n` },
    { path: ".claude/skills/tejo/SKILL.md", content: skill(folder) },
    { path: "AGENTS.md", content: withBlock(readIfExists(join(cwd, "AGENTS.md")), agentsBlock(folder)) },
  ];
  const claude = readIfExists(join(cwd, "CLAUDE.md"));
  if (claude !== null && !claude.includes("@AGENTS.md")) {
    writes.push({ path: "CLAUDE.md", content: `${claude.replace(/\n*$/, "\n")}\n@AGENTS.md\n` });
  }

  const report: InitReport = { created: [], updated: [], unchanged: [] };
  const folderPath = join(cwd, folder);
  if (hasDocuments(folderPath)) report.unchanged.push(`${folder}/`);
  else writes.push({ path: `${folder}/bienvenida.md`, content: WELCOME_NOTE });

  for (const { path, content } of writes) {
    const file = join(cwd, path);
    const current = readIfExists(file);
    if (current === content) {
      report.unchanged.push(path);
      continue;
    }
    mkdirSync(dirname(file), { recursive: true });
    writeFileSync(file, content);
    (current === null ? report.created : report.updated).push(path);
  }
  return report;
}
```

export const DEFAULT_PORT = 8356;
const MAX_BODY = 5 * 1024 * 1024;
const VIEW_FILE = /^([A-Za-z0-9_-]+)\.json$/;

const MIME: Record<string, string> = {
  ".html": "text/html; charset=utf-8",
  ".js": "text/javascript; charset=utf-8",
  ".css": "text/css; charset=utf-8",
  ".svg": "image/svg+xml",
  ".png": "image/png",
  ".json": "application/json; charset=utf-8",
  ".woff2": "font/woff2",
  ".ico": "image/x-icon",

function send(res: ServerResponse, status: number, body: unknown): void {
  res.writeHead(status, { "content-type": "application/json; charset=utf-8" });
  res.end(JSON.stringify(body));
}

/** Anti DNS-rebinding y CSRF (#66): solo el propio origen local. */
function isLocal(req: IncomingMessage, port: number): boolean {
  const allowed = [`127.0.0.1:${port}`, `localhost:${port}`];
  if (!allowed.includes(req.headers.host ?? "")) return false;
  const origin = req.headers.origin;
  return origin === undefined || allowed.some((host) => origin === `http://${host}`);
}

function readBody(req: IncomingMessage): Promise<string> {
  return new Promise((resolve, reject) => {
    const chunks: Buffer[] = [];
    let size = 0;
    req.on("data", (chunk: Buffer) => {
      size += chunk.length;
      if (size > MAX_BODY) reject(new Error("el archivo supera 5 MB"));
      else chunks.push(chunk);
    });
    req.on("end", () => resolve(Buffer.concat(chunks).toString("utf8")));
    req.on("error", reject);
  });
}

/** `.flo/views/*.json` de la carpeta, sin seguir enlaces simbólicos. */
function readViews(root: string): Record<string, string> {
  const dir = join(root, ".flo", "views");
  const views: Record<string, string> = {};
  for (const part of [join(root, ".flo"), dir]) {
    if (!lstatSync(part, { throwIfNoEntry: false })?.isDirectory()) return views;
  }
  for (const name of readdirSync(dir)) {
    const match = VIEW_FILE.exec(name);
    const file = join(dir, name);
    if (match && lstatSync(file).isFile()) views[match[1]] = readFileSync(file, "utf8");
  }
  return views;
}

/** Estáticos del visor; null si la ruta sale de `viewerDir` o no existe. */
function staticFile(viewerDir: string, pathname: string): string | null {
  let decoded: string;
  try {
    decoded = decodeURIComponent(pathname);
  } catch {
    return null;
  }
  const file = normalize(join(viewerDir, decoded === "/" ? "index.html" : decoded));
  if (!file.startsWith(normalize(viewerDir + sep))) return null;
  return statSync(file, { throwIfNoEntry: false })?.isFile() ? file : null;
}

/** El servidor de `tejo view` (spec §4.3, decisión #66). */
export function createViewServer(folder: FloFolder, viewerDir: string): Server {
  const clients = new Set<ServerResponse>();
  let watcher: { close(): void } | null = null;

  const server = createServer((req, res) => {
    void handle(req, res).catch((error) => send(res, 500, { error: (error as Error).message }));
  });

  async function handle(req: IncomingMessage, res: ServerResponse): Promise<void> {
    const { port } = server.address() as AddressInfo;
    const url = new URL(req.url ?? "/", "http://localhost");
    if (!url.pathname.startsWith("/api/")) {
      const file = req.method === "GET" ? staticFile(viewerDir, url.pathname) : null;
      if (!file) return send(res, 404, { error: "no encontrado" });
      res.writeHead(200, { "content-type": MIME[extname(file)] ?? "application/octet-stream" });
      res.end(readFileSync(file));
      return;
    }
    if (!isLocal(req, port)) return send(res, 403, { error: "solo se aceptan pedidos desde el propio visor local" });

    if (req.method === "GET" && url.pathname === "/api/files") {
      return send(res, 200, { root: basename(folder.root), files: folder.files() });
    }
    if (req.method === "GET" && url.pathname === "/api/views") return send(res, 200, readViews(folder.root));
    if (req.method === "GET" && url.pathname === "/api/events") {
      res.writeHead(200, { "content-type": "text/event-stream", "cache-control": "no-cache", connection: "keep-alive" });
      res.write(": tejo\n\n");
      clients.add(res);
      req.on("close", () => clients.delete(res));
      return;
    }
    if (req.method === "PUT" && url.pathname.startsWith("/api/files/")) {
      let uri: string;
      try {
        uri = url.pathname.slice("/api/files/".length).split("/").map(decodeURIComponent).join("/");
      } catch {
        return send(res, 400, { error: "ruta mal codificada" });
      }
      // Una barra invertida o un segmento '..' nunca es una uri del workspace.
      if (uri.includes("\\") || uri.split("/").some((part) => part === ".." || part === "")) {
        return send(res, 400, { error: `ruta inválida: '${uri}'` });
      }
      const result = folder.save(uri, await readBody(req));
      return result.ok ? send(res, 200, { uri: result.uri }) : send(res, 400, { error: result.error });
    }
    send(res, 404, { error: "no encontrado" });
  }

  server.on("listening", () => {
    watcher = folder.watch(() => {
      for (const client of clients) client.write("event: change\ndata: {}\n\n");
    });
  });
  server.on("close", () => {
    watcher?.close();
    for (const client of clients) client.end();
    clients.clear();
  });
  return server;
}

function openBrowser(url: string): void {
  const [command, args] =
    process.platform === "darwin" ? ["open", [url]] : process.platform === "win32" ? ["cmd", ["/c", "start", "", url]] : ["xdg-open", [url]];
  // ponytail: sin navegador (servidor, CI) solo queda la URL impresa.
  spawn(command, args, { stdio: "ignore", detached: true }).on("error", () => {}).unref();
}

/** `tejo view`: carga la carpeta, escucha en 127.0.0.1 y abre el navegador. */
export async function startView(
  root: string,
  options: { port: number; open: boolean; viewerDir: string }
): Promise<{ url: string; server: Server }> {
  if (!existsSync(join(options.viewerDir, "index.html"))) {
    throw new Error(`no se encontró el visor en '${options.viewerDir}' (¿falta el build?)`);
  }
  const folder = new FloFolder(root);
  folder.load();
  const server = createViewServer(folder, options.viewerDir);
  await new Promise<void>((resolve, reject) => {
    server.once("error", (error: NodeJS.ErrnoException) =>
      reject(error.code === "EADDRINUSE" ? new Error(`el puerto ${options.port} está ocupado; prueba con --port <otro>`) : error)
    );
    server.listen(options.port, "127.0.0.1", resolve);
  });
  const { port } = server.address() as AddressInfo;
  const url = `http://127.0.0.1:${port}/`;
  if (options.open) openBrowser(url);
  return { url, server };
}

import type { WorkspaceSnapshot } from "./workspace.js";

/** La API de `tejo view` (decisión #66), relativa a la página. */
const encode = (uri: string) => uri.split("/").map(encodeURIComponent).join("/");

async function getJson(path: string): Promise<unknown> {
  const res = await fetch(path, { headers: { accept: "application/json" } });
  if (!res.ok || !(res.headers.get("content-type") ?? "").includes("application/json")) {
    throw new Error(`sin API local en ${path}`);
  }
  return res.json();
}

export async function readLocalFiles(): Promise<Record<string, string>> {
  const body = (await getJson("api/files")) as { files?: unknown };
  if (typeof body.files !== "object" || body.files === null) throw new Error("respuesta inválida de api/files");
  return body.files as Record<string, string>;
}

/** Con `tejo view` detrás, el workspace es la carpeta del disco; sin él (Pages, vite), null. */
export async function openLocalFolder(): Promise<WorkspaceSnapshot | null> {
  let files: Record<string, string>;
  try {
    files = await readLocalFiles();
  } catch {
    return null;
  }
  const uris = Object.keys(files).sort();
  return { files, saved: { ...files }, active: uris[0] ?? "", dir: null, local: true };
}

export async function writeLocalFile(uri: string, source: string): Promise<void> {
  const res = await fetch(`api/files/${encode(uri)}`, { method: "PUT", body: source });
  if (!res.ok) {

```ts
  /** Con carpeta (File System Access o `tejo view`): "Nuevo" agrega y "Guardar" escribe todo lo modificado. */
  hasFolder(): boolean {
    return this.dir !== null || this.local;
  }

  /**
   * Aplica el estado del disco (modo local, #66). Lo limpio sigue al disco:
   * se actualiza, se agrega o se quita. Lo modificado en el editor se
   * conserva y, si el disco lo cambió o borró, es un conflicto. Lo nunca
   * guardado que no está en disco no se toca.
   */
  applyDisk(files: Record<DocumentUri, string>): { changed: DocumentUri[]; conflicts: DocumentUri[] } {
    const changed: DocumentUri[] = [];
    const conflicts: DocumentUri[] = [];
    const uris = new Set([...this.files.keys(), ...Object.keys(files)]);
    for (const uri of [...uris].sort()) {
      const disk = files[uri];
      const saved = this.saved.get(uri);
      if (disk === saved) continue;
      if (this.isDirty(uri)) {
        conflicts.push(uri);
        if (disk !== undefined) this.saved.set(uri, disk);
        else this.saved.delete(uri);
        continue;
      }
      if (disk === undefined) {
        this.files.delete(uri);
        this.saved.delete(uri);
        this.core.removeDocument(uri);
        this.linked = null;
      } else {
        this.setSource(uri, disk);
        this.saved.set(uri, disk);
      }
      changed.push(uri);
    }
    if (!this.files.has(this.active)) this.active = this.uris()[0] ?? "";
    return { changed, conflicts };
  }
```
